        private void ShowUserMove(Button ObjectButton)
        {
            ObjectButton.Text = mFlag;
            mArrayMoves[ObjectButton.TabIndex] = 1;
            mMoveNumber++;
            mFlag = "0";
        }

        private void ShowNPCMove(List<Button> ListButtons, int index)
        {
            ListButtons[index].Text = mFlag;
            mArrayMoves[index] = 3;
            mMoveNumber++;
            mFlag = "X";
        }

        private int GenerateRandomNPCMove(List<Button> ListButtons)
        {
            int index = 0;
            Random RandomNumber = new Random();

            do
            {
                index = RandomNumber.Next(0, 9);
                if (mArrayMoves[index] == 0 && index < mArrayMoves.Length)
                    return index;
            }
            while (true);
        }

        public bool PlayUser(Button ObjectButton, List<Button> ListButtons,
                             GroupBox grbTicTacToe)
        {
            if (ObjectButton.Text == "")

                    mGameControl = CheckWinner(ListButtons, grbTicTacToe);

        public void PlayNPC(List<Button> ListButtons, GroupBox grbTicTacToe)
        {
            int index;
            Random NumeroRandomico = new Random();

            if (mGameControl == 0)
            {
                if (mMoveNumber == 1)
                {
                    if (mArrayMoves[4] == 0)
                    {
                        index = 4;
                        ShowNPCMove(ListButtons, index);
                        mGameControl = CheckWinner(ListButtons, grbTicTacToe);
                        return;
                    }
                    do
                    {
                        index = NumeroRandomico.Next(0, 9);
                        if (mArrayMoves[index] == 0 && (index == 0 || index == 2 || index == 6 || index == 8))
                        {
                            ShowNPCMove(ListButtons, index);
                            mGameControl = CheckWinner(ListButtons, grbTicTacToe);
                            return;
                        }
                    } while (true);
                }
                else // mMoveNumber != 1
                {
                    //if (mArrayMoves[4] == 0)
                    //{
                    //    index = 4;
                    //    ShowNPCMove(ListButtons, index);
                    //    mGameControl = CheckWinner(ListButtons, grbTicTacToe);
                    //    return;
                    //}
                    //else
                    //{
                        index = HeuristicNumericalMethod(ListButtons);

                        if (mArrayMoves[4] == 3 && mSum == 5)
                        {
                            do
                            {
                                index = NumeroRandomico.Next(0, 9);
                                if (mArrayMoves[index] == 0 && (index == 1 || index == 3 || index == 5 || index == 7))
                                {
                                    ShowNPCMove(ListButtons, index);
                                    mGameControl = CheckWinner(ListButtons, grbTicTacToe);
                                    return;
                                }
                            } while (true);
                        }

                        ShowNPCMove(ListButtons, index);
                        mGameControl = CheckWinner(ListButtons, grbTicTacToe);
                        return;

            else
                return;
